\subsection{Definitions}
% conventions: time scales, frames, software terms used on the following slides
\begin{frame}{Conventions and definitions}
  \scriptsize
  \renewcommand{\arraystretch}{1.05}
  \begin{tabular}{@{}l >{\raggedright\arraybackslash}p{0.88\textwidth}@{}}
    \grp{Time scales}
    UTC & civil time; kept within \SI{0.9}{s} of the Earth's rotation by leap seconds \\
    TAI, TT & International Atomic Time: TAI $-$ UTC $=$ \SI{37}{s} (since 2017); Terrestrial Time $=$ TAI $+$ \SI{32.184}{s} \\
    TDB & Barycentric Dynamical Time, of the JPL ephemerides and SPICE (``ET''); TDB $-$ UTC $=$ \SI{69.18}{s} (2026) \\
    UT1 & the Earth's actual rotation (measured by the IERS): sets where the site is \\
    TCB & Barycentric Coordinate Time: Gaia's reference epoch J2016.0 is given in TCB \\[0.3ex]
    \grp{Reference frames}
    ICRF & International Celestial Reference Frame (quasars): Gaia, DE440, Horizons; SPICE's ``J2000'' frame \\
    ITRF & International Terrestrial Reference Frame (realizes the ITRS): fixed to the rotating Earth (ITRF93) \\
    WGS84 & the GPS ellipsoid: the site's latitude, longitude and height (EGM96 geoid: height above sea level) \\[0.3ex]
    \grp{Software and data}
    SPICE & NAIF's (JPL) toolkit for ephemerides and orientation; SPK: position files, PCK: orientation files \\
    DE440 & JPL's planetary ephemeris (Sun, planets, Moon) \\
    ``CN'' & SPICE's converged Newtonian light-time correction \\
    ERFA & Essential Routines for Fundamental Astronomy: open copy of the IAU's SOFA library (astropy) \\
  \end{tabular}
\end{frame}

\subsection{Astrometry}
\begin{frame}{Astrometry: the star from the Gaia catalog to the event}\slideid{ST1}
  \footnotesize\tightdisplays\setbeamerfont{footnote}{size=\tiny}
  \setlength{\leftmargini}{1.4em}
  \begin{enumerate}\setlength{\itemsep}{0.2ex}
    \item \OWN\ \textbf{Gaia DR3} at epoch J2016.0 (ICRF, barycentric):
          position $\alpha, \delta$; parallax $\varpi$ (mas; distance $1000/\varpi$ pc);
          proper motion $\mu_{\alpha*} = \mu_\alpha \cos\delta$, $\mu_\delta$ (mas/yr); magnitude $G$;
          radial velocity $v_r$ (km/s, fast nearby stars).\\
          2-parameter stars (position only): proper motion from UCAC5 / UCAC4 or UCAC4's epoch to Gaia
    \item \OWN\ \textbf{Bright-star frame spin} ($G < 13$, Gaia's own proper motions only):\\
          Cantat-Gaudin \& Brandt 2021\footnote{T.~Cantat-Gaudin, T.~D.~Brandt, Characterizing and correcting the
          proper motion bias of the bright Gaia EDR3 sources, A\&A 649, A124 (2021);
          \href{https://doi.org/10.1051/0004-6361/202140807}{doi:10.1051/0004-6361/202140807}},
          $\omega$ in \si{\micro as/yr} per $G$ bin, up to 80
          \[
            \tcbhighmath{\begin{aligned}
              \mu_{\alpha*} &\leftarrow \mu_{\alpha*} - \tfrac{1}{1000}\,(-\sin\delta\cos\alpha\,\omega_x
                               - \sin\delta\sin\alpha\,\omega_y + \cos\delta\,\omega_z)\\
              \mu_\delta &\leftarrow \mu_\delta - \tfrac{1}{1000}\,(\sin\alpha\,\omega_x - \cos\alpha\,\omega_y)
            \end{aligned}}
            \tag{ST1.1}
          \]
    \item \AP\ \textbf{Space motion} from J2016.0 to the event (straight line, incl.\ light time)\\
          \texttt{SkyCoord(\ldots).apply\_space\_motion(t)} from 2016-01-01T12:00 TCB $\to$ ERFA \texttt{pmsafe}; distance
          $1000/\max(\varpi, 0.01)$\,pc, $v_r$ only for fast nearby stars (else 0)
          \[
            \tcbhighmath{\uv{p} = \mathrm{pmsafe}(\alpha, \delta, \mu_{\alpha*}, \mu_\delta, \varpi, v_r;\ \text{J2016.0} \to t)}
            \tag{ST1.2}
          \]
  \end{enumerate}
\end{frame}


\begin{frame}{Astrometry: the star seen from the Earth}\slideid{ST2}
  \footnotesize\tightdisplays\setbeamerfont{footnote}{size=\tiny}
  \setlength{\leftmargini}{1.4em}
  \begin{enumerate}\setlength{\itemsep}{0.4ex}\setcounter{enumi}{3}
    \item \OWN\ \textbf{Stellar parallax} (if $\varpi > 0$), seen from the Earth's center\\
          \SP\ $\vv{E}$ = Earth from the barycenter (geometric)
          \[
            \tcbhighmath{\uv{p} \leftarrow \frac{\uv{p}\,(1000/\varpi)\,\mathrm{pc} - \vv{E}}{|\uv{p}\,(1000/\varpi)\,\mathrm{pc} - \vv{E}|}}
            \tag{ST2.1}
          \]
    \item \OWN\ \textbf{Light deflection} by the Sun (\SI{4}{mas} at \SI{90}{\degree}, $1.75''$ at the limb), Jupiter, Saturn
          (SOFA \texttt{iauLd} form\footnote{S.~A.~Klioner, A practical relativistic model for microarcsecond astrometry in
          space, AJ 125, 1580--1597 (2003); \href{https://doi.org/10.1086/367593}{doi:10.1086/367593}}) --- \emph{star minus asteroid}
          (the asteroid's light is bent too)\\
          \SP\ bodies and asteroid from the Earth's center (``CN''): $\uv{e}_b$, $E_b$ body $\to$ Earth's center,
          $\uv{q}_b$ body $\to$ asteroid, $GM_b$ from DE440
          \[
            \tcbhighmath{\delta\uv{p} = \sum_{b} \frac{2GM_b}{c^2 E_b}\Bigl[\vv{f}(\uv{p}) - \vv{f}(\uv{q}_b)\Bigr],
            \qquad
            \vv{f}(\uv{q}) = \frac{\uv{e}_b\,(\uv{p}\cdot\uv{q}) - \uv{q}\,(\uv{p}\cdot\uv{e}_b)}{1 + \uv{q}\cdot\uv{e}_b}}
            \tag{ST2.2}
          \]
          \[
            \tcbhighmath{\uv{s} = \frac{\uv{p} + \delta\uv{p}}{|\uv{p} + \delta\uv{p}|}}
            \tag{ST2.3}
          \]
  \end{enumerate}
\end{frame}

% astrometry.corrected_star_dir (deflection star minus asteroid), paths.path_error (Horizons RSS 3-sigma).
\begin{frame}{Astrometry: the asteroid}\slideid{AS}
  \footnotesize\tightdisplays
  \setlength{\leftmargini}{1.4em}
  \textbf{Propagation --- two stages}
  \begin{itemize}
    \item \OWN\ \textbf{Screening} (pick, pre-screens: all asteroids at once): SBDB osculating elements
          (full precision), own RK4 with Sun + planets (\SP\ DE440): $\approx 0.01''$ vs.\ Horizons
    \item \SP\ \textbf{Exact prediction}: JPL Horizons orbit per asteroid as an SPK file
          (JPL's integration), refreshed after a week, kept for offline use
  \end{itemize}
  \smallskip\textbf{Corrections}
  \begin{itemize}
    \item \SP\ \textbf{Light time} from \emph{the observer} (\texttt{spkcpo}), not the Earth's center:
          $\vv{\rho} = \vv{A}(t-\tau) - \vv{O}(t)$ (LT.2)\\
          \OWN\ path lines: each ground point $\vv{G}$ its own light time, the center line shifted by
          \[
            \tcbhighmath{\vv{v}_{\perp}\,\bigl(|\vv{A} - \vv{E}| - |\vv{A} - \vv{G}|\bigr)/c}
            \tag{AS.1}
          \]
  \end{itemize}
\end{frame}

\begin{frame}{Astrometry: the asteroid (continued)}
  \footnotesize
  \setlength{\leftmargini}{1.4em}
  \textbf{Corrections}
  \begin{itemize}\setlength{\itemsep}{1ex}
    \item \textbf{No aberration} applied: it shifts star and asteroid alike (astrometric, as Gaia)
    \item \OWN\ \textbf{Light deflection} of the asteroid's own light: subtracted from the star's
          (the $\vv{f}(\uv{q}_b)$ term), so the asteroid stays astrometric
    \item \textbf{Uncertainty}: Horizons 3-$\sigma$ error ellipse (RSS) $\to$ path $\sigma$,
          combined with the star's position error
  \end{itemize}
\end{frame}

\subsection{Solver}

\begin{frame}{The exact solver: inputs}\slideid{IN}
  \footnotesize
  \begin{itemize}
    \item \textbf{Star:} unit vector $\uv{s}$ in ICRF (J2000), \emph{corrected}: Gaia DR3 propagated to the event
          (proper motion, parallax, radial velocity), stellar parallax from the Earth, light deflection by
          Sun, Jupiter, Saturn --- computed once at the guessed time $t_0$:
          \[
            \tcbhighmath{\uv{s} = (\cos\delta\cos\alpha,\ \cos\delta\sin\alpha,\ \sin\delta)^{\mathsf T}}
            \tag{IN.1}
          \]
    \item \textbf{Asteroid:} barycentric position $\vv{A}(t)$ from its JPL Horizons orbit file (SPK type 21, about the
          Sun; \SP\ adds the Sun from DE440): the current orbit solution (``JPL\#$n$'', full-precision osculating
          elements fitted to all astrometry), integrated with the planets, Moon, 16 most massive asteroids, relativity,
          the Earth's oblateness and non-gravitational forces. Uncertainty: Horizons' 3-$\sigma$ (RSS) $\to$ path
          $\sigma$; the size: next slide
    \item \textbf{Observer:} geodetic $(\varphi, \lambda, h)$ on WGS84, $\lambda$ east-positive, $h$ above the ellipsoid
          ($=$ height above sea level $+$ EGM96 geoid height)
    \item \SP\ \textbf{Time:} $t$ = TDB (SPICE ``ET''); UTC $\to$ TDB with
          the leap seconds (\texttt{str2et}); Earth rotation (UT1, polar motion) from NAIF's Earth orientation file
  \end{itemize}
\end{frame}

% the asteroid's size (search.get_asteroid_size): the order of the sources, and the size from H and albedo
\begin{frame}{The asteroid's size}\slideid{SZ}
  \footnotesize\tightdisplays
  The shadow radius $r = D/2$ decides the hit test (CA.2) and the duration (OV.2); its range ($r_{\min}$, $r_{\max}$)
  widens the search (CO.1). First found wins:
  \begin{enumerate}\setlength{\itemsep}{0.3ex}
    \item A curated value (\texttt{SIZE\_OVERRIDES}), if set
    \item \textbf{Measured diameter} from JPL SBDB (IR surveys, occultations, radar: the reference is kept)
          $D \pm \sigma$ ($\sigma$ missing: 15\,\%); range $D \pm 3\sigma$, widened by a known tri-axial extent
    \item \textbf{No diameter:} from the absolute magnitude $H$ and the geometric albedo $p$\footnote{P.~Pravec,
          A.~W.~Harris, Binary asteroid population. 1. Angular momentum content, Icarus 190, 250--259 (2007),
          appendix A; \href{https://doi.org/10.1016/j.icarus.2007.02.023}{doi:10.1016/j.icarus.2007.02.023}}
          \[
            \tcbhighmath{D = \frac{1329\,\mathrm{km}}{\sqrt{p}}\;10^{-H/5}}
            \tag{SZ.1}
          \]
          SBDB albedo known: range $p \times 1.5$ \ldots\ $p / 1.5$;\quad
          none: $p = 0.14$ assumed, range $0.30$ \ldots\ $0.05$ (diameter uncertain by a factor $\approx 1.7$:
          marked \emph{size est.} in the report)
    \item Neither diameter nor $H$: no size, the asteroid is skipped
  \end{enumerate}
\end{frame}

\begin{frame}{Step 1: the observer in space}\slideid{OB}
  \SP\ WGS84: $a = \SI{6378.137}{km}$, $f = 1/298.257223563$, $e^2 = f(2-f)$,
  $N = a / \sqrt{1 - e^2 \sin^2\varphi}$
  \[
    \tcbhighmath{\vv{r}_{\mathrm{ITRF}} =
    \begin{pmatrix}
      (N + h)\cos\varphi\cos\lambda \\
      (N + h)\cos\varphi\sin\lambda \\
      \bigl(N(1 - e^2) + h\bigr)\sin\varphi
    \end{pmatrix}}
    \tag{OB.1}
  \]
  \SP\ \texttt{georec(lon, lat, h, a, f)} (WGS84 constants from our \texttt{earth.py})\\[0.5ex]
  Rotate to the celestial frame and add the Earth's barycentric position (inside \texttt{spkcpo}):
  \[
    \tcbhighmath{\vv{O}(t) = \vv{E}(t) + \mathsf{R}(t)\,\vv{r}_{\mathrm{ITRF}}}
    \tag{OB.2}
  \]
  {\small \SP\ $\mathsf{R}(t) = $ \texttt{pxform('ITRF93', 'J2000', t)} from NAIF's binary Earth orientation file \texttt{earth\_latest\_high\_prec.bpc}
  (before 2000: \texttt{earth\_1962\_\ldots\_combined.bpc}): precession, nutation, UT1, polar motion}
\end{frame}

\begin{frame}{Step 2: the asteroid as seen from the site}\slideid{LT}
  \SP\ \texttt{spkcpo(ast, t, 'J2000', 'OBSERVER', 'CN', r\_ITRF, '399', 'ITRF93')}:
  light time $\tau$ from the asteroid to \emph{this observer} (converged Newtonian):
  \begin{gather*}
    \tcbhighmath{c\,\tau = \bigl|\vv{A}(t - \tau) - \vv{O}(t)\bigr|} \tag{LT.1} \\
    \tcbhighmath{\vv{\rho}(t) = \vv{A}(t - \tau) - \vv{O}(t)} \tag{LT.2}
  \end{gather*}
  \begin{itemize}
    \item Astrometric (``CN''), no stellar aberration: it shifts star and asteroid alike
    \item Light time from the site, not from the Earth's center: the site is up to
          $R_\oplus \sin h_\star / c \approx \SI{21}{ms}$ closer
          $\Rightarrow$ up to $\approx\SI{0.6}{km}$ on the ground
          {\small(found comparing with Occult)}
  \end{itemize}
\end{frame}

\begin{frame}{Step 3: the observer's fundamental plane}\slideid{FP}
  \OWN\ Plane through the observer, perpendicular to the star:
  \[
    \tcbhighmath{\uv{z} = \uv{s},\qquad
    \uv{x} = \frac{\uv{e}_3 \times \uv{s}}{|\uv{e}_3 \times \uv{s}|}\ \text{(east)},\qquad
    \uv{y} = \uv{s} \times \uv{x}\ \text{(north)}}
    \tag{FP.1}
  \]
  Shadow axis relative to the observer, and the miss distance:
  \[
    \tcbhighmath{\xi(t) = \uv{x}\cdot\vv{\rho}(t),\qquad
    \eta(t) = \uv{y}\cdot\vv{\rho}(t),\qquad
    d(t) = \sqrt{\xi(t)^2 + \eta(t)^2}}
    \tag{FP.2}
  \]
  {\small The star is at infinity: its shadow is a cylinder of radius $r$ along $\uv{s}$;
  the observer is inside it when $d(t) < r$.}\\[1ex]
  {\footnotesize\color{black!75}\textbf{Not the classical (Besselian) plane:} that one goes through the Earth's center, with
  the light time from the Earth's center, and each observer is projected onto it. Ours goes through the observer,
  with the light time from the site (LT): the same geometry, no projection and no $z/c$ term.}
\end{frame}

\begin{frame}{Step 4: closest approach and the hit test}\slideid{CA}
  Minimise over an offset from the guess $t_0$ (bracket $T$ from the corridor scan):
  \[
    \tcbhighmath{t^{*} = t_0 + \operatorname*{arg\,min}_{|\Delta t| \le T/2} d(t_0 + \Delta t),
    \qquad d_{\min} = d(t^{*})}
    \tag{CA.1}
  \]
  \begin{itemize}
    \item \SC\ \texttt{minimize\_scalar(method='bounded')} (Brent), \texttt{xatol} \SI{1}{ms}; solving for $\Delta t$, not raw ET:
          for $t \approx 8\times10^{8}$\,s the tolerance $\sqrt{\epsilon}\,|t|$ would be $\approx\SI{12}{s}$
    \item Minimum within \SI{1}{s} of the bracket edge: not a real minimum (a neighbour finds it)
  \end{itemize}
  \[
    \tcbhighmath{\text{hit} \iff d_{\min} < r_{\max} + \text{reach}
    \qquad(\text{logged ``in the shadow'': } d_{\min} < r_{\max})}
    \tag{CA.2}
  \]
\end{frame}

\begin{frame}{Step 5: observable?}\slideid{OV}
  \OWN\ Local vertical (geodetic normal) and altitudes at $t^{*}$; \SP\ $\mathsf{R}$ by \texttt{pxform},
  Sun $\uv{s}_\odot$ from the Earth's center, apparent (``LT+S''):
  \[
    \tcbhighmath{\begin{aligned}
      \uv{u} &= \mathsf{R}(t^{*})
        \begin{pmatrix}\cos\varphi\cos\lambda\\ \cos\varphi\sin\lambda\\ \sin\varphi\end{pmatrix}\\
      h_\star &= \arcsin(\uv{u}\cdot\uv{s}) > h_{\min},
      \qquad
      h_\odot = \arcsin(\uv{u}\cdot\uv{s}_\odot) < h_{\odot,\max}
    \end{aligned}}
    \tag{OV.1}
  \]
  {\small Both at the observer's closest approach $t^{*}$; the star limit $h_{\min}$ and the Sun limit $h_{\odot,\max}$
  are the site's (sites.py).}
\end{frame}

\begin{frame}{Step 5: what the observer sees}\slideid{OV}
  Speed, chord, duration and drop:
  \begin{gather*}
    \tcbhighmath{v = \frac{\bigl|(\xi,\eta)(t^{*}{+}1\,\mathrm{s}) - (\xi,\eta)(t^{*}{-}1\,\mathrm{s})\bigr|}{2\,\mathrm{s}},
    \qquad
    \ell = 2\sqrt{r^2 - d_{\min}^2},
    \qquad
    \Delta t_{\mathrm{occ}} = \frac{\ell}{v}} \tag{OV.2} \\
    \tcbhighmath{\Delta m = 2.5\log_{10}\!\bigl(1 + 10^{0.4(m_{\mathrm{ast}} - m_\star)}\bigr)} \tag{OV.3}
  \end{gather*}
  {\small \OWN\ $m_{\mathrm{ast}}$ from $H$, $G$ (IAU H--G system; \SP\ distances, phase angle); kept if $\Delta m \ge 0.1$}
\end{frame}
